Estrutura SQL (Executar no Supabase SQL Editor): 
create table if not exists portfolio (
  id text primary key,
  ticker text not null,
  name text not null,
  category text not null,
  broker text not null,
  qty numeric not null,
  "avgUSD" numeric not null,
  "currUSD" numeric not null,
  "feesUSD" numeric default 0,
  updated_at timestamp with time zone default timezone('utc'::text, now())
);

alter table portfolio enable row level security;
create policy "Permitir tudo" on portfolio for all using (true) with check (true);
alter publication supabase_realtime add table portfolio;
